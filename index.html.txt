    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Элементы DOM
            const elements = {
                lengthSlider: document.getElementById('length-slider'),
                lengthValue: document.getElementById('length-value'),
                uppercase: document.getElementById('uppercase'),
                lowercase: document.getElementById('lowercase'),
                numbers: document.getElementById('numbers'),
                symbols: document.getElementById('symbols'),
                generateBtn: document.getElementById('generate-btn'),
                copyBtn: document.getElementById('copy-btn'),
                password: document.getElementById('password'),
                strengthText: document.getElementById('strength-text'),
                strengthFill: document.getElementById('strength-fill'),
                hackTime: document.getElementById('hack-time'),
                notification: document.getElementById('notification')
            };
            
            // Наборы символов
            const charsets = {
                uppercase: 'ABCDEFGHIJKLMNOPQRSTUVWXYZ',
                lowercase: 'abcdefghijklmnopqrstuvwxyz',
                numbers: '0123456789',
                symbols: '!@#$%^&*()_+-=[]{}|;:,.<>?'
            };
            
            // Обновление слайдера
            elements.lengthSlider.addEventListener('input', function() {
                elements.lengthValue.textContent = this.value;
            });
            
            // Генерация пароля
            elements.generateBtn.addEventListener('click', generatePassword);
            
            // Копирование пароля
            elements.copyBtn.addEventListener('click', copyPassword);
            
            // Автоматическая генерация при загрузке
            generatePassword();
            
            // Функция генерации пароля
            function generatePassword() {
                const length = parseInt(elements.lengthSlider.value);
                let charset = '';
                
                // Собираем набор символов
                if (elements.uppercase.checked) charset += charsets.uppercase;
                if (elements.lowercase.checked) charset += charsets.lowercase;
                if (elements.numbers.checked) charset += charsets.numbers;
                if (elements.symbols.checked) charset += charsets.symbols;
                
                // Проверка, что хотя бы один тип выбран
                if (charset === '') {
                    elements.password.textContent = '⚠️ Выберите хотя бы один тип символов!';
                    return;
                }
                
                // Генерация криптографически безопасного пароля
                let password = '';
                const randomValues = new Uint32Array(length);
                window.crypto.getRandomValues(randomValues);
                
                for (let i = 0; i < length; i++) {
                    const randomIndex = randomValues[i] % charset.length;
                    password += charset[randomIndex];
                }
                
                // Отображаем пароль
                elements.password.textContent = password;
                
                // Обновляем оценку надёжности
                updatePasswordStrength(password);
            }
            
            // Оценка надёжности пароля
            function updatePasswordStrength(password) {
                let score = 0;
                const length = password.length;
                
                // Баллы за длину
                if (length >= 8) score += 1;
                if (length >= 12) score += 2;
                if (length >= 16) score += 3;
                if (length >= 20) score += 4;
                
                // Баллы за разнообразие
                if (/[A-Z]/.test(password)) score += 2;
                if (/[a-z]/.test(password)) score += 2;
                if (/[0-9]/.test(password)) score += 2;
                if (/[^A-Za-z0-9]/.test(password)) score += 3;
                
                // Уровень надёжности
                let strength, color, percent;
                if (score >= 12) {
                    strength = 'ОТЛИЧНО 🔒';
                    color = '#2e7d32';
                    percent = 100;
                } else if (score >= 9) {
                    strength = 'ОЧЕНЬ СИЛЬНЫЙ 🔐';
                    color = '#388e3c';
                    percent = 85;
                } else if (score >= 6) {
                    strength = 'СИЛЬНЫЙ 👍';
                    color = '#f57c00';
                    percent = 65;
                } else if (score >= 3) {
                    strength = 'СРЕДНИЙ ⚠️';
                    color = '#ff9800';
                    percent = 45;
                } else {
                    strength = 'СЛАБЫЙ ❌';
                    color = '#d32f2f';
                    percent = 25;
                }
                
                // Обновление интерфейса
                elements.strengthText.textContent = strength;
                elements.strengthText.style.color = color;
                elements.strengthFill.style.width = percent + '%';
                elements.strengthFill.style.background = color;
                
                // Расчёт времени взлома
                updateHackTime(password);
            }
            
            // Расчёт времени взлома
            function updateHackTime(password) {
                const charsetSize = calculateCharsetSize(password);
                const length = password.length;
                
                // Количество возможных комбинаций
                const combinations = Math.pow(charsetSize, length);
                
                // Предполагаемая скорость взлома (попыток в секунду)
                const guessesPerSecond = 1e9; // 1 миллиард попыток/сек
                
                // Время в секундах
                const seconds = combinations / guessesPerSecond;
                
                // Форматирование времени
                let timeText;
                if (seconds < 1) {
                    timeText = 'менее секунды';
                } else if (seconds < 60) {
                    timeText = Math.round(seconds) + ' секунд';
                } else if (seconds < 3600) {
                    const minutes = Math.round(seconds / 60);
                    timeText = minutes + ' минут' + (minutes === 1 ? 'у' : '');
                } else if (seconds < 86400) {
                    const hours = Math.round(seconds / 3600);
                    timeText = hours + ' час' + (hours === 1 ? '' : hours < 5 ? 'а' : 'ов');
                } else if (seconds < 31536000) {
                    const days = Math.round(seconds / 86400);
                    timeText = days + ' день' + (days === 1 ? '' : days < 5 ? 'я' : 'ей');
                } else {
                    const years = Math.round(seconds / 31536000);
                    timeText = years + ' год' + (years === 1 ? '' : years < 5 ? 'а' : 'ов');
                }
                
                elements.hackTime.textContent = `⏱️ Взлом за: ~${timeText} (при 1 млрд попыток/сек)`;
            }
            
            // Подсчёт размера набора символов
            function calculateCharsetSize(password) {
                let size = 0;
                if (/[A-Z]/.test(password)) size += 26;
                if (/[a-z]/.test(password)) size += 26;
                if (/[0-9]/.test(password)) size += 10;
                if (/[^A-Za-z0-9]/.test(password)) size += 20;
                return size > 0 ? size : 1;
            }
            
            // Копирование пароля в буфер обмена
            function copyPassword() {
                const password = elements.password.textContent;
                
                // Проверка, что пароль есть
                if (password.includes('Выберите') || password.includes('Нажмите')) {
                    return;
                }
                
                // Используем современный API
                navigator.clipboard.writeText(password)
                    .then(() => {
                        showNotification();
                    })
                    .catch(err => {
                        // Запасной вариант для старых браузеров
                        const textArea = document.createElement('textarea');
                        textArea.value = password;
                        document.body.appendChild(textArea);
                        textArea.select();
                        document.execCommand('copy');
                        document.body.removeChild(textArea);
                        showNotification();
                    });
            }
            
            // Показать уведомление
            function showNotification() {
                elements.notification.classList.add('show');
                setTimeout(() => {
                    elements.notification.classList.remove('show');
                }, 2000);
            }
            
            // Автоматическая генерация при изменении настроек
            ['uppercase', 'lowercase', 'numbers', 'symbols', 'length-slider'].forEach(id => {
                document.getElementById(id).addEventListener('change', generatePassword);
            });
        });
    </script>
